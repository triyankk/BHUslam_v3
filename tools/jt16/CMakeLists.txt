project(bhu_jt16_tools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(jt16_inspect_feed inspect_feed.cpp)
target_include_directories(jt16_inspect_feed SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../tof_bridge/cpp/vendor")
set_target_properties(jt16_inspect_feed PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(BHU_JT16_MAPPING "Build C++ ROS2 adapter for existing Hesai FAST-LIO runtime" OFF)
if(BHU_JT16_MAPPING)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  add_executable(jt16_map_bridge map_bridge.cpp)
  target_include_directories(jt16_map_bridge PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
  target_include_directories(jt16_map_bridge SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../tof_bridge/cpp/vendor")
  ament_target_dependencies(jt16_map_bridge rclcpp sensor_msgs nav_msgs)
  set_target_properties(jt16_map_bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
