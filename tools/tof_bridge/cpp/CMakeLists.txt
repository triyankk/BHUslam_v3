cmake_minimum_required(VERSION 3.16)
project(tof_native_viewer LANGUAGES CXX)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(tof_viewer main.cpp)
target_compile_features(tof_viewer PRIVATE cxx_std_17)
target_compile_options(tof_viewer PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(tof_viewer SYSTEM PRIVATE vendor)
target_include_directories(tof_viewer PRIVATE ../../../include)
target_compile_definitions(tof_viewer PRIVATE TOF_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(tof_viewer PRIVATE Threads::Threads OpenSSL::Crypto)
enable_testing()
add_test(NAME native_protocol COMMAND tof_viewer --self-test)
