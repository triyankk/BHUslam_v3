cmake_minimum_required(VERSION 3.16)
project(bhu_vins_fusion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
get_filename_component(BHU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VINS_ROOT "${BHU_ROOT}/third_party/VINS-Fusion")
set(VINS_DEPS "${BHU_ROOT}/third_party/vins-deps")
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(Ceres 1.14 REQUIRED PATHS "${VINS_DEPS}/install/lib/cmake/Ceres" NO_DEFAULT_PATH)
set(CV_INCLUDE "${BHU_ROOT}/third_party/sysroot/usr/include/opencv4" /usr/include/opencv4)
set(CMAKE_FIND_LIBRARY_SUFFIXES .so .so.4.5d)
foreach(c core imgproc imgcodecs calib3d features2d video videoio highgui flann)
  unset(CV_${c} CACHE)
  find_library(CV_${c} NAMES opencv_${c} PATHS "${BHU_ROOT}/third_party/sysroot/usr/lib/aarch64-linux-gnu" REQUIRED)
  list(APPEND CV_LIBS "${CV_${c}}")
endforeach()
set(CAM "${VINS_ROOT}/camera_models")
file(GLOB CAM_SOURCES "${CAM}/src/camera_models/*.cc")
add_library(fusion_camera STATIC ${CAM_SOURCES} "${CAM}/src/gpl/gpl.cc" "${CAM}/src/gpl/EigenQuaternionParameterization.cc")
target_include_directories(fusion_camera SYSTEM PUBLIC "${CAM}/include" "${BHU_ROOT}/third_party/sysroot/usr/include" ${CV_INCLUDE} ${CERES_INCLUDE_DIRS})
target_link_libraries(fusion_camera PUBLIC Eigen3::Eigen ${CV_LIBS} ${CERES_LIBRARIES})
set(CORE "${VINS_ROOT}/vins_estimator/src")
# Only substitute the ROS visualization include in a build-directory copy.
file(READ "${CORE}/estimator/estimator.cpp" ESTIMATOR_SOURCE)
string(REPLACE "#include \"../utility/visualization.h\"" "#include \"${CMAKE_CURRENT_SOURCE_DIR}/visualization_stub.hpp\"" ESTIMATOR_SOURCE "${ESTIMATOR_SOURCE}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/estimator_no_ros.cpp" CONTENT "${ESTIMATOR_SOURCE}")
file(READ "${CORE}/featureTracker/feature_tracker.cpp" TRACKER_SOURCE)
set(REVERSE_LK "cv::calcOpticalFlowPyrLK(rightImg, cur_img, cur_right_pts, reverseLeftPts, statusRightLeft, err, cv::Size(21, 21), 3);")
string(FIND "${TRACKER_SOURCE}" "${REVERSE_LK}" REVERSE_POSITION)
if(REVERSE_POSITION EQUAL -1)
  message(FATAL_ERROR "Upstream stereo tracker changed; review the rectified policy patch")
endif()
string(REPLACE "${REVERSE_LK}" "if(bhu::fusion::rectified_stereo) { reverseLeftPts=cur_pts; cv::calcOpticalFlowPyrLK(rightImg, cur_img, cur_right_pts, reverseLeftPts, statusRightLeft, err, cv::Size(21,21), 3, cv::TermCriteria(cv::TermCriteria::COUNT+cv::TermCriteria::EPS,30,0.01), cv::OPTFLOW_USE_INITIAL_FLOW); } else { ${REVERSE_LK} }" TRACKER_SOURCE "${TRACKER_SOURCE}")
string(REPLACE "distance(cur_pts[i], reverseLeftPts[i]) <= 0.5)" "distance(cur_pts[i], reverseLeftPts[i]) <= 0.5 && bhu::fusion::stereo_geometry(cur_pts[i],cur_right_pts[i]))" TRACKER_SOURCE "${TRACKER_SOURCE}")
string(PREPEND TRACKER_SOURCE "#include \"${CMAKE_CURRENT_SOURCE_DIR}/stereo_policy.hpp\"\n")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/feature_tracker_rectified.cpp" CONTENT "${TRACKER_SOURCE}")
file(GLOB FACTORS "${CORE}/factor/*.cpp")
file(GLOB INITIAL "${CORE}/initial/*.cpp")
add_library(fusion_core STATIC "${CMAKE_BINARY_DIR}/estimator_no_ros.cpp" "${CORE}/estimator/feature_manager.cpp" "${CMAKE_BINARY_DIR}/feature_tracker_rectified.cpp" "${CORE}/utility/utility.cpp" ${FACTORS} ${INITIAL})
target_include_directories(fusion_core PRIVATE "${CORE}/featureTracker")
target_include_directories(fusion_core PUBLIC "${CORE}/estimator" "${CMAKE_CURRENT_SOURCE_DIR}/shim" "${BHU_ROOT}/tools/vins_mono/shim")
target_link_libraries(fusion_core PUBLIC fusion_camera pthread)
add_executable(vins_fusion_replay main.cpp)
target_include_directories(vins_fusion_replay PRIVATE "${BHU_ROOT}/include")
target_include_directories(vins_fusion_replay SYSTEM PRIVATE "${BHU_ROOT}/tools/tof_bridge/cpp/vendor")
target_link_libraries(vins_fusion_replay PRIVATE fusion_core)
find_library(FUSION_ORBBEC NAMES OrbbecSDK PATHS "${BHU_ROOT}/third_party/orbbec/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(vins_fusion_live live.cpp)
target_include_directories(vins_fusion_live PRIVATE "${BHU_ROOT}/include" "${BHU_ROOT}/third_party/orbbec/include")
target_include_directories(vins_fusion_live SYSTEM PRIVATE "${BHU_ROOT}/tools/tof_bridge/cpp/vendor")
target_link_libraries(vins_fusion_live PRIVATE fusion_core "${FUSION_ORBBEC}")
target_compile_options(vins_fusion_live PRIVATE -O1 -Wno-deprecated-declarations)
foreach(t fusion_camera fusion_core vins_fusion_replay)
  target_compile_options(${t} PRIVATE -O1 -Wno-deprecated-declarations)
endforeach()
