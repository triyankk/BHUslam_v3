cmake_minimum_required(VERSION 3.16)
project(bhu_livo_offline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(BHU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LIVO "${BHU_ROOT}/third_party/FAST-LIVO2")
set(VKIT "${BHU_ROOT}/third_party/fast-livo-vikit/vikit_common")
set(SOPHUS "${BHU_ROOT}/third_party/fast-livo-sophus")
set(RUNTIME "/home/atas/vscode/optFlow_slam/runtime/lio")
include_directories(SYSTEM /usr/include/opencv4 /usr/include/eigen3 "${RUNTIME}/pcl/include/pcl-1.12" "${RUNTIME}/sysroot/usr/include" "${RUNTIME}/sysroot/usr/include/eigen3" "${BHU_ROOT}/third_party/sysroot/usr/include/opencv4" "${BHU_ROOT}/third_party/sysroot/usr/include")
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/shim" "${LIVO}/include" "${VKIT}/include" "${SOPHUS}")
add_compile_definitions(ROOT_DIR="${CMAKE_BINARY_DIR}/" MP_PROC_NUM=1 ARM_ARCH)
add_compile_options(-O1 -fopenmp -Wno-deprecated-declarations)
# Preserve all upstream numerical source files. Only old Eigen API compatibility
# is changed in a build-directory copy of Sophus.
file(READ "${SOPHUS}/sophus/so2.cpp" SO2)
string(REPLACE "unit_complex_.real() = 1.;" "unit_complex_.real(1.);" SO2 "${SO2}")
string(REPLACE "unit_complex_.imag() = 0.;" "unit_complex_.imag(0.);" SO2 "${SO2}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/so2.cpp" CONTENT "${SO2}")
include_directories("${SOPHUS}/sophus")
add_library(livo_sophus STATIC "${CMAKE_BINARY_DIR}/so2.cpp" "${SOPHUS}/sophus/so3.cpp" "${SOPHUS}/sophus/se3.cpp")
add_library(livo_camera STATIC "${VKIT}/src/pinhole_camera.cpp" "${VKIT}/src/math_utils.cpp" "${VKIT}/src/vision.cpp" "${VKIT}/src/robust_cost.cpp" "${VKIT}/src/performance_monitor.cpp")
file(READ "${LIVO}/src/IMU_Processing.cpp" IMU_SOURCE)
# Set the propagation time reference after initialization. The original leaves
# last_prop_end_time uninitialized before its first numerical propagation.
string(REPLACE "init_iter_num = 1;" "init_iter_num = 1; last_prop_end_time = 0.; time_last_scan = 0.;" IMU_SOURCE "${IMU_SOURCE}")
string(REPLACE "last_imu = meas.imu.back();\n\n    if" "last_imu = meas.imu.back();\n    last_prop_end_time = pcl_end_time;\n    lidar_meas.last_lio_update_time = pcl_end_time;\n\n    if" IMU_SOURCE "${IMU_SOURCE}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/imu_offline.cpp" CONTENT "${IMU_SOURCE}")
add_library(livo_core STATIC "${LIVO}/src/vio.cpp" "${LIVO}/src/voxel_map.cpp" "${CMAKE_BINARY_DIR}/imu_offline.cpp" "${LIVO}/src/frame.cpp" "${LIVO}/src/visual_point.cpp")
set(CV_LIBS)
foreach(c core imgproc imgcodecs calib3d features2d video highgui)
  find_library(CV_${c} NAMES opencv_${c} PATHS "${BHU_ROOT}/third_party/sysroot/usr/lib/aarch64-linux-gnu" REQUIRED)
  list(APPEND CV_LIBS "${CV_${c}}")
endforeach()
target_link_libraries(livo_camera PUBLIC livo_sophus ${CV_LIBS})
target_link_libraries(livo_core PUBLIC livo_camera "${RUNTIME}/pcl/lib/libpcl_filters.so" "${RUNTIME}/pcl/lib/libpcl_common.so" gomp pthread)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/replay.cpp")
  add_executable(livo_replay replay.cpp)
  target_link_libraries(livo_replay PRIVATE livo_core)
endif()
enable_testing()
add_executable(livo_shim_test shim_test.cpp)
add_test(NAME offline_time_rotation_publication_guard COMMAND livo_shim_test)
