cmake_minimum_required(VERSION 3.16)
project(bhu_vins_mono LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
get_filename_component(BHU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VINS_ROOT "${BHU_ROOT}/third_party/VINS-Mono")
set(VINS_DEPS "${BHU_ROOT}/third_party/vins-deps")
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(Ceres 1.14 REQUIRED PATHS "${VINS_DEPS}/install/lib/cmake/Ceres" NO_DEFAULT_PATH)
set(CV_INCLUDE "${BHU_ROOT}/third_party/sysroot/usr/include/opencv4" /usr/include/opencv4)
set(CMAKE_FIND_LIBRARY_SUFFIXES .so .so.4.5d)
foreach(c core imgproc imgcodecs calib3d features2d video videoio highgui flann)
  unset(CV_${c} CACHE)
  find_library(CV_${c} NAMES opencv_${c} PATHS "${BHU_ROOT}/third_party/sysroot/usr/lib/aarch64-linux-gnu" REQUIRED)
  list(APPEND CV_LIBS "${CV_${c}}")
endforeach()
set(CAM "${VINS_ROOT}/camera_model")
file(GLOB CAM_SOURCES "${CAM}/src/camera_models/*.cc")
add_library(vins_camera STATIC ${CAM_SOURCES} "${CAM}/src/gpl/gpl.cc" "${CAM}/src/gpl/EigenQuaternionParameterization.cc")
target_include_directories(vins_camera SYSTEM PUBLIC "${CAM}/include" "${BHU_ROOT}/third_party/sysroot/usr/include" ${CV_INCLUDE} ${CERES_INCLUDE_DIRS})
target_link_libraries(vins_camera PUBLIC Eigen3::Eigen ${CV_LIBS} ${CERES_LIBRARIES})
add_library(vins_frontend STATIC frontend.cpp "${VINS_ROOT}/feature_tracker/src/feature_tracker.cpp")
target_include_directories(vins_frontend PRIVATE "${VINS_ROOT}/feature_tracker/src" shim)
target_compile_definitions(vins_frontend PRIVATE ROW=TRACKER_ROW COL=TRACKER_COL FOCAL_LENGTH=TRACKER_FOCAL_LENGTH WINDOW_SIZE=TRACKER_WINDOW_SIZE NUM_OF_CAM=TRACKER_NUM_OF_CAM)
target_link_libraries(vins_frontend PUBLIC vins_camera)
set(CORE "${VINS_ROOT}/vins_estimator/src")
file(GLOB FACTORS "${CORE}/factor/*.cpp")
file(GLOB INITIAL "${CORE}/initial/*.cpp")
add_library(vins_core STATIC "${CORE}/estimator.cpp" "${CORE}/feature_manager.cpp" "${CORE}/utility/utility.cpp" ${FACTORS} ${INITIAL})
target_include_directories(vins_core PUBLIC "${CORE}" shim)
target_link_libraries(vins_core PUBLIC vins_camera pthread)
add_executable(vins_mono_replay main.cpp)
target_include_directories(vins_mono_replay PRIVATE "${BHU_ROOT}/include")
target_include_directories(vins_mono_replay SYSTEM PRIVATE "${BHU_ROOT}/tools/tof_bridge/cpp/vendor")
target_link_libraries(vins_mono_replay PRIVATE vins_frontend vins_core)
add_executable(vins_mono_report report.cpp)
target_include_directories(vins_mono_report SYSTEM PRIVATE "${BHU_ROOT}/tools/tof_bridge/cpp/vendor" ${CV_INCLUDE})
target_link_libraries(vins_mono_report PRIVATE ${CV_LIBS})
target_compile_definitions(vins_mono_report PRIVATE BHU_VINS_PAGE="${BHU_ROOT}/web/vins-compare.html")
enable_testing()
add_executable(vins_integration_test integration_test.cpp)
target_include_directories(vins_integration_test PRIVATE "${CORE}" shim)
target_link_libraries(vins_integration_test PRIVATE vins_camera)
target_compile_options(vins_integration_test PRIVATE -O1)
add_test(NAME vins_native_integration COMMAND vins_integration_test)
# Bounded Eigen compilation on the Jetson. Both libraries use the same compiler.
foreach(t vins_camera vins_frontend vins_core vins_mono_replay)
  target_compile_options(${t} PRIVATE -O1 -Wno-deprecated-declarations)
endforeach()
