if(BHU_BUILD_SLAM)
  find_package(Eigen3 3.1 REQUIRED NO_MODULE)
  find_package(Pangolin REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_library(BOOST_SERIALIZATION boost_serialization PATHS "${BHU_SYSROOT}/lib/aarch64-linux-gnu" REQUIRED)
  file(GLOB DBOW_SOURCES "${ORB_ROOT}/Thirdparty/DBoW2/DBoW2/*.cpp" "${ORB_ROOT}/Thirdparty/DBoW2/DUtils/*.cpp")
  add_library(DBoW2 SHARED ${DBOW_SOURCES})
  target_include_directories(DBoW2 PUBLIC ${BHU_CV_INCLUDE})
  target_link_libraries(DBoW2 PUBLIC ${BHU_CV_LIBS})
  # Build upstream's bundled optimizer without upstream's global compiler flags.
  set(g2o_SOURCE_DIR "${ORB_ROOT}/Thirdparty/g2o")
  set(G2O_SHARED_LIBS ON)
  configure_file("${g2o_SOURCE_DIR}/config.h.in" "${g2o_SOURCE_DIR}/config.h")
  file(READ "${g2o_SOURCE_DIR}/CMakeLists.txt" G2O_CMAKE)
  string(REGEX MATCHALL "g2o/[A-Za-z0-9_/]+\\.(cpp|c)" G2O_REL_SOURCES "${G2O_CMAKE}")
  set(G2O_SOURCES "")
  foreach(source ${G2O_REL_SOURCES})
    string(STRIP "${source}" source)
    list(APPEND G2O_SOURCES "${g2o_SOURCE_DIR}/${source}")
  endforeach()
  add_library(g2o SHARED ${G2O_SOURCES})
  target_include_directories(g2o PUBLIC "${g2o_SOURCE_DIR}" "${g2o_SOURCE_DIR}/g2o/core" "${g2o_SOURCE_DIR}/g2o/stuff" "${g2o_SOURCE_DIR}/g2o/types")
  target_link_libraries(g2o PUBLIC Eigen3::Eigen)
  file(GLOB ORB_SOURCES "${ORB_ROOT}/src/*.cc" "${ORB_ROOT}/src/*.cpp" "${ORB_ROOT}/src/CameraModels/*.cpp")
  add_library(ORB_SLAM3 SHARED ${ORB_SOURCES})
  # GCC's -O3 optimization of these Eigen-heavy files exhausts this 8 GB host
  # when paired with other work. Keep their compilation bounded and serial.
  set_source_files_properties(
    "${ORB_ROOT}/src/Frame.cc" "${ORB_ROOT}/src/G2oTypes.cc"
    "${ORB_ROOT}/src/Optimizer.cc" "${ORB_ROOT}/src/Tracking.cc"
    "${ORB_ROOT}/src/LocalMapping.cc" "${ORB_ROOT}/src/LoopClosing.cc"
    PROPERTIES COMPILE_OPTIONS "-O1")
  target_compile_definitions(ORB_SLAM3 PUBLIC COMPILEDWITHC11)
  target_include_directories(ORB_SLAM3 PUBLIC "${ORB_ROOT}" "${ORB_ROOT}/include"
    "${ORB_ROOT}/include/CameraModels" "${ORB_ROOT}/Thirdparty/Sophus"
    "${CMAKE_SOURCE_DIR}/third_party/Pangolin/components/pango_video/include" ${BHU_CV_INCLUDE})
  target_link_libraries(ORB_SLAM3 PUBLIC DBoW2 g2o Eigen3::Eigen pango_display pango_plot
    ${BHU_CV_LIBS} ${BOOST_SERIALIZATION} OpenSSL::Crypto pthread)
  add_executable(gemini_replay src/replay.cpp)
  target_include_directories(gemini_replay PRIVATE include)
  target_include_directories(gemini_replay SYSTEM PRIVATE "${CMAKE_SOURCE_DIR}/tools/tof_bridge/cpp/vendor")
  target_compile_definitions(gemini_replay PRIVATE BHU_SPARSE_PAGE="${CMAKE_SOURCE_DIR}/web/sparse-map.html")
  target_link_libraries(gemini_replay PRIVATE ORB_SLAM3)
  if(BHU_BUILD_CAMERA AND BHU_BUILD_EXTNAV)
    add_executable(gemini_live src/live.cpp)
    target_include_directories(gemini_live PRIVATE include)
    target_include_directories(gemini_live SYSTEM PRIVATE "${BHU_MAVLINK_VENDOR}")
    target_link_libraries(gemini_live PRIVATE ORB_SLAM3 orbbec_sdk bhu_benchmark)
  endif()
endif()
